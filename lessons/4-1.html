Most-used SQL commands

Replace the names in capitals with your own. Each block copies with one click.

Warehouses

-- Create, resize, suspend, pick
CREATE WAREHOUSE ANALYTICS_WH WITH WAREHOUSE_SIZE = 'XSMALL' AUTO_SUSPEND = 60 AUTO_RESUME = TRUE;
ALTER WAREHOUSE ANALYTICS_WH SET WAREHOUSE_SIZE = 'MEDIUM';   -- scale up
ALTER WAREHOUSE ANALYTICS_WH SET MAX_CLUSTER_COUNT = 3;       -- scale out (Enterprise)
ALTER WAREHOUSE ANALYTICS_WH SUSPEND;
USE WAREHOUSE ANALYTICS_WH;
CREATE RESOURCE MONITOR MONTHLY_CAP WITH CREDIT_QUOTA = 100
  TRIGGERS ON 80 PERCENT DO NOTIFY ON 100 PERCENT DO SUSPEND;

Objects

CREATE DATABASE BANK_DB;
CREATE SCHEMA BANK_DB.RAW;
CREATE TABLE BANK_DB.RAW.PAYMENTS (id NUMBER, amount NUMBER(18,2), paid_at TIMESTAMP_NTZ, details VARIANT);
CREATE TRANSIENT TABLE BANK_DB.RAW.STAGING_LOAD (id NUMBER);   -- no Fail-safe
CREATE TEMPORARY TABLE SCRATCH (id NUMBER);                    -- gone when session ends
CREATE VIEW BANK_DB.RAW.BIG_PAYMENTS AS SELECT * FROM BANK_DB.RAW.PAYMENTS WHERE amount > 10000;
USE SCHEMA BANK_DB.RAW;
SHOW TABLES; DESCRIBE TABLE PAYMENTS;

Loading

CREATE FILE FORMAT CSV_FMT TYPE = CSV SKIP_HEADER = 1 FIELD_OPTIONALLY_ENCLOSED_BY = '"';
CREATE STAGE PAYMENTS_STAGE FILE_FORMAT = CSV_FMT;
LIST @PAYMENTS_STAGE;
COPY INTO PAYMENTS FROM @PAYMENTS_STAGE VALIDATION_MODE = RETURN_ERRORS;  -- test first
COPY INTO PAYMENTS FROM @PAYMENTS_STAGE ON_ERROR = 'ABORT_STATEMENT';
CREATE PIPE PAYMENTS_PIPE AUTO_INGEST = TRUE AS
  COPY INTO PAYMENTS FROM @PAYMENTS_STAGE;                        -- Snowpipe

Time Travel

SELECT * FROM PAYMENTS AT (OFFSET => -3600);                -- one hour ago
SELECT * FROM PAYMENTS AT (TIMESTAMP => '2026-09-01 09:00:00'::TIMESTAMP_NTZ);
SELECT * FROM PAYMENTS BEFORE (STATEMENT => '<query-id>');     -- just before that query
UNDROP TABLE PAYMENTS;
ALTER TABLE PAYMENTS SET DATA_RETENTION_TIME_IN_DAYS = 30;

Cloning

CREATE TABLE PAYMENTS_TEST CLONE PAYMENTS;
CREATE DATABASE BANK_DB_DEV CLONE BANK_DB;
CREATE TABLE PAYMENTS_YESTERDAY CLONE PAYMENTS AT (OFFSET => -86400);  -- clone + Time Travel

Sharing

CREATE SECURE VIEW BANK_DB.RAW.PAYMENTS_SUMMARY AS SELECT city, SUM(amount) total FROM PAYMENTS GROUP BY city;
CREATE SHARE PARTNER_SHARE;
GRANT USAGE ON DATABASE BANK_DB TO SHARE PARTNER_SHARE;
GRANT USAGE ON SCHEMA BANK_DB.RAW TO SHARE PARTNER_SHARE;
GRANT SELECT ON VIEW BANK_DB.RAW.PAYMENTS_SUMMARY TO SHARE PARTNER_SHARE;
ALTER SHARE PARTNER_SHARE ADD ACCOUNTS = <org_name>.<account_name>;

Roles and security

CREATE ROLE ANALYST;
GRANT USAGE ON WAREHOUSE ANALYTICS_WH TO ROLE ANALYST;
GRANT USAGE ON DATABASE BANK_DB TO ROLE ANALYST;
GRANT USAGE ON SCHEMA BANK_DB.RAW TO ROLE ANALYST;
GRANT SELECT ON ALL TABLES IN SCHEMA BANK_DB.RAW TO ROLE ANALYST;
GRANT ROLE ANALYST TO USER PRIYA;
USE ROLE ANALYST;
CREATE MASKING POLICY MASK_CARD AS (val STRING) RETURNS STRING ->
  CASE WHEN CURRENT_ROLE() IN ('FRAUD_ANALYST') THEN val ELSE '****' END;
ALTER TABLE PAYMENTS MODIFY COLUMN card_number SET MASKING POLICY MASK_CARD;
CREATE ROW ACCESS POLICY REGION_ONLY AS (region STRING) RETURNS BOOLEAN ->
  CURRENT_ROLE() = 'GLOBAL_RISK' OR region = 'IN';
ALTER TABLE PAYMENTS ADD ROW ACCESS POLICY REGION_ONLY ON (region);

Streams, tasks and dynamic tables

CREATE STREAM PAYMENTS_STREAM ON TABLE PAYMENTS;
SELECT * FROM PAYMENTS_STREAM;                     -- METADATA$ACTION, METADATA$ISUPDATE
CREATE TASK LOAD_SUMMARY WAREHOUSE = ANALYTICS_WH SCHEDULE = '10 MINUTES'
  WHEN SYSTEM$STREAM_HAS_DATA('PAYMENTS_STREAM')
  AS INSERT INTO PAYMENTS_SUMMARY SELECT city, SUM(amount) FROM PAYMENTS_STREAM GROUP BY city;
ALTER TASK LOAD_SUMMARY RESUME;                      -- new tasks start suspended
CREATE DYNAMIC TABLE PAYMENTS_DAILY TARGET_LAG = '1 hour' WAREHOUSE = ANALYTICS_WH
  AS SELECT paid_at::DATE d, SUM(amount) total FROM PAYMENTS GROUP BY d;

Cortex AI

SELECT AI_COMPLETE('mistral-large', 'Explain a wire transfer in one sentence');
SELECT AI_CLASSIFY(complaint_text, ['fees', 'fraud', 'app bug']) FROM COMPLAINTS;
SELECT AI_SENTIMENT(review_text) FROM REVIEWS;
SELECT branch, AI_SUMMARIZE_AGG(complaint_text) FROM COMPLAINTS GROUP BY branch;
SELECT AI_TRANSLATE(complaint_text, 'en', 'fr') FROM COMPLAINTS;
-- Cortex Code CLI (in your terminal, not SQL): cortex   |   cortex -p "explain this query"

Confusing pairs

PairFirst oneSecond oneHow to keep them apart
Time Travel vs Fail-safeYou use it yourself with SQL, 1 to 90 daysSnowflake Support uses it for you, 7 days after Time Travel endsRewind the CCTV yourself; the vault needs a guard
Scale up vs scale outBigger warehouse, one heavy query runs fasterMore clusters, many users stop queuingSlow query: up. Long queue: out
Clone vs shareYour own writable copy in your accountRead-only live view of the original for another accountPhotocopy you can scribble on vs a glass window
Stream vs taskRecords what changedRuns SQL on a scheduleThe notebook does not act; the alarm clock does not record
Cortex AI functions vs Cortex CodeSQL functions that run an LLM over rows of dataAn assistant that writes and fixes the SQL itselfSmart reader reads the books; robot writes the request slips
Transient vs temporaryStays until dropped, visible to others, no Fail-safeVanishes when your session ends, only you see itBoth skip Fail-safe; only temporary dies with the session
Permanent vs transientUp to 90 days Time Travel plus 7 days Fail-safeUp to 1 day Time Travel, no Fail-safe, cheaperReal bank data: permanent. Staging and scratch: transient
Result cache vs warehouse cacheSaved answer to the exact same query, 24 hours, no warehouse neededData files kept on a running warehouse's local diskManager's memory vs the librarian's desk pile
COPY INTO vs SnowpipeYou run it, on your warehouse, in bulkRuns by itself as files land, on Snowflake computeManual trolley vs conveyor belt
Masking vs row access policyHides a column valueHides whole rowsTape over a line vs a locked page
Cortex Analyst vs Cortex SearchPlain question in, SQL over structured tables outFinds matching text passages by meaningAnalyst counts the books; Search finds the paragraph
